\section{The SPSA ``process''}

We analyze an elegant chess–oriented version of the SPSA algorithm invented by
Stockfish author Joona Kiiski.

Let $(\bm{c}_i)_{i \ge 0} \subset \mathbb{R}^n$ be a sequence of column vectors and let $(r_i)_{i \ge 0} \subset \mathbb{R}_{>0}$. To
go from a vector of engine parameters


\[
  \bm{\theta}_i = (\theta_i^{(j)})_{j=1}^n \in \mathbb{R}^n
\]


to a new vector $\bm{\theta}_{i+1} \in \mathbb{R}^n$ we play a match of $\bm{\theta}_i + \bm{c}_i$ against $\bm{\theta}_i - \bm{c}_i$ and put
\begin{equation}
  \bm{\theta}_{i+1} =
  \begin{cases}
    \bm{\theta}_i + r_i \bm{c}_i & \text{in case of a win}, \

\[4pt]
    \bm{\theta}_i                & \text{in case of a draw}, \

\[4pt]
    \bm{\theta}_i - r_i \bm{c}_i & \text{in case of a loss.}
  \end{cases}
  \label{eq:update}
\end{equation}

Since we have biased\footnote{We assume games are played with a perfectly balanced opening book. In practice this is approximated by replaying each game with reversed colors.}
the update of $\bm{\theta}_i$ in the direction of the engine that won the
game, the hope is that $\bm{\theta}_i$ for $i > 0$ will yield a stronger engine than $\bm{\theta}_0$.

The next two examples illustrate standard choices for $\bm{c}_i$.

\begin{example}
Let $c_0,\dots,c_{n-1} > 0$ and put


\[
  \bm{c}_i = c_{i \bmod n} \,\bm{e}_{i \bmod n},
\]


where $\bm{e}_j$ is the $j$th standard basis vector of $\mathbb{R}^n$. This is a variant of the FDSA algorithm. We have


\[
  \mathbb{E}(\bm{c}_i \bm{c}_i^\top) = \operatorname{diag}(c_0^2,\dots,c_{n-1}^2).
\]


\end{example}

\begin{example}
Let $c_0,\dots,c_{n-1} > 0$ and put


\[
  \bm{c}_i = \sum_{j=0}^{n-1} \alpha_{i,j} c_j \bm{e}_j,
\]


where each $\alpha_{i,j}$ is chosen uniformly at random in $\{\pm 1\}$. Then


\[
  \mathbb{E}(\bm{c}_i \bm{c}_i^\top) = \operatorname{diag}(c_0^2,\dots,c_{n-1}^2).
\]


This is the SPSA algorithm.
\end{example}

\begin{assumption}
We assume $(\bm{c}_i \bm{c}_i^\top)_i$ stays close to its short–term running average
$\mathbb{E}(\bm{c}_i \bm{c}_i^\top)$ and that this matrix is invertible.
\end{assumption}

\section{Main results}

\subsection{Statements}

Let $e$ map $\bm{\theta} \in \mathbb{R}^n$ to Elo. Assume $e$ is approximately quadratic and attains its maximum at $\bm{\theta}^\ast$, normalized so that $e(\bm{\theta}^\ast)=0$.

Apply generalized SPSA with constant $r$ and with $\bm{c}_i$ satisfying Assumption~\ref{ass:Ecct}.

\begin{assumption}
\label{ass:Ecct}
$\mathbb{E}(\bm{c}_i \bm{c}_i^\top)$ exists, is constant in $i$, and is invertible.
\end{assumption}

\begin{enumerate}
\item Asymptotically,
\begin{equation}
  \mathbb{E}\bigl(e(\bm{\theta}_i)\bigr)
  \approx
  - \frac{r}{8}\, C\, n (1-d),
  \label{eq:main1}
\end{equation}
where $d$ is the draw ratio and


\[
  C = \frac{800}{\ln 10} \approx 347.43558552.
\]



\item Let $0 \le \gamma \le 1$ and define $z$ by $\gamma = \mathbb{P}(X < z)$ for $X \sim \chi^2_n$.
Then with probability $\gamma$,
\begin{equation}
  e(\bm{\theta}_i)
  \gtrsim
  - \frac{r}{8}\, C\, z (1-d).
  \label{eq:main2}
\end{equation}

\item The expectation $\mathbb{E}(\bm{\theta}_i)$ converges linearly to $\bm{\theta}^\ast$ with time constant
\begin{equation}
  \lambda \approx \frac{C}{2 r \mu},
  \label{eq:lambda}
\end{equation}
where $\mu$ is the smallest eigenvalue of


\[
  \mathbb{E}(\bm{c}_i \bm{c}_i^\top)\, \operatorname{Hess}(e)(\bm{\theta}^\ast).
\]


\end{enumerate}

\begin{remark}
The ideal case is


\[
  \mathbb{E}(\bm{c}_i \bm{c}_i^\top)\, \operatorname{Hess}(e)(\bm{\theta}^\ast) = \mu I_n,
\]


in which the update~\eqref{eq:update} is (on average)


\[
  r\, \operatorname{Hess}(e)(\bm{\theta}^\ast)^{-1} (\nabla e)(\bm{\theta}_i),
\]


the optimal gradient–descent direction.
\end{remark}

\subsection{Derivations}

Below vectors are always column vectors. We switch to continuous time and write $\bm{\theta}(t)$ instead of $\bm{\theta}_i$.

We can think of the dynamics of $\bm{\theta}$ as a discrete version of the continuous process
\begin{equation}
  d\bm{\theta}_t
  = r \bm{c}\,\bigl(\mu_{\theta,\bm{c}}\,dt + \sigma_{\theta,\bm{c}}\,dW_t\bigr),
\end{equation}
where $(\mu_{\theta,\bm{c}}, \sigma_{\theta,\bm{c}})$ are the mean and standard deviation of a single game between
parameters $\bm{\theta} \pm \bm{c}$ (scored as $-1,0,1$) and $W_t$ is a Wiener process. We assume $\sigma_{\theta,\bm{c}}$ is independent of $\bm{\theta}, \bm{c}$ so that
\begin{equation}
  \sigma_{\theta,\bm{c}}^2 = 1 - d,
  \label{eq:sigma}
\end{equation}
where $d$ is the draw ratio.

Let $f(\bm{\theta})$ be the function that maps $\bm{\theta}$ to the expected score against a reference
engine (scored as $-1,0,1$). Assuming linearity we get
\begin{align*}
  \mu_{\theta,\bm{c}}
  &= \operatorname{score}(\bm{\theta}+\bm{c},\bm{\theta}-\bm{c}) \\
  &= f(\bm{\theta}+\bm{c}) - f(\bm{\theta}-\bm{c})
   = 2 \bm{c}^\top (\nabla f)(\bm{\theta}),
\end{align*}
and hence
\begin{equation}
  d\bm{\theta}_t
  = 2 r \bm{c}\bm{c}^\top (\nabla f)(\bm{\theta})\,dt
    + r \sigma_{\theta,\bm{c}} \bm{c}\,dW_t.
\end{equation}

We now replace $\bm{c}\bm{c}^\top$ by its short–term average / expectation value $\mathbb{E}(\bm{c}\bm{c}^\top)$ and set


\[
  A = r\,\mathbb{E}(\bm{c}\bm{c}^\top).
\]


Note that $A$ is symmetric. Then
\begin{equation}
  d\bm{\theta}_t
  = 2 A (\nabla f)(\bm{\theta})\,dt + r \sigma_{\theta,\bm{c}} \bm{c}\,dW_t.
\end{equation}

The conversion between $f$ and $e$ is approximately
\begin{equation}
  f(\bm{\theta}) = \frac{e(\bm{\theta})}{C}.
  \label{eq:f-e}
\end{equation}
So $f$ is approximately quadratic. Translating so that $\bm{\theta}^\ast = \bm{0}$, we may write


\[
  f(\bm{\theta}) = -\tfrac{1}{2} \bm{\theta}^\top E \bm{\theta},
\]


with $E$ a positive–definite symmetric $n \times n$ matrix. Thus


\[
  (\nabla f)(\bm{\theta}) = - E \bm{\theta},
\]


and
\begin{equation}
  d\bm{\theta}_t
  = -2 A E \bm{\theta}\,dt + r \sigma_{\theta,\bm{c}} \bm{c}\,dW_t.
\end{equation}

The corresponding homogeneous equation


\[
  d\bm{\theta}_t = -2 A E \bm{\theta}\,dt
\]


has solution


\[
  \bm{\theta}(t) = \exp(-2 A E t)\,\bm{D},
\]


where $\bm{D}$ is constant. Using variation of constants (letting $\bm{D}$ depend on $t$), we get


\[
  \exp(-2 A E t)\, d\bm{D}(t) = r \sigma_{\theta,\bm{c}} \bm{c}\,dW_t,
\]


hence


\[
  \bm{D}(t)
  = \bm{D}(0)
    + r \sigma_{\theta,\bm{c}} \int_0^t \exp(2 A E s)\,\bm{c}\,dW_s.
\]


So
\begin{equation}
  \bm{\theta}(t)
  = \exp(-2 A E t)\,\bm{\theta}(0)
    + r \sigma_{\theta,\bm{c}} \int_0^t \exp(2 A E (s-t))\,\bm{c}\,dW_s.
\end{equation}

Thus $\bm{\theta}(t)$ is multivariate normal with expectation


\[
  \mathbb{E}(\bm{\theta}(t))
  = \exp(-2 A E t)\,\bm{\theta}(0)
  = \exp\bigl(-2 r\,\mathbb{E}(\bm{c}\bm{c}^\top) E\,t\bigr)\,\bm{\theta}(0),
\]


from which the convergence in (3) follows, using~\eqref{eq:f-e} and~\eqref{eq:sigma}.

Now for the covariance. Using the stochastic integral term and again replacing $\bm{c}\bm{c}^\top$ by its
average, we obtain
\begin{align*}
  \operatorname{Cov}(\bm{\theta}(t))
  &= r^2 \sigma_{\theta,\bm{c}}^2
     \int_0^t \exp(2 A E (s-t))\,\bm{c}\bm{c}^\top\,\exp(2 E A (s-t))\,ds \\
  &\approx r^2 \sigma_{\theta,\bm{c}}^2
     \int_0^t \exp(2 A E (s-t))\,\mathbb{E}(\bm{c}\bm{c}^\top)\,\exp(2 E A (s-t))\,ds \\
  &= r^2 \sigma_{\theta,\bm{c}}^2
     \int_0^t \exp\bigl(4 E A (s-t)\bigr)\,A\,ds.
\end{align*}

Assuming $E$ and $A$ commute, we get


\[
  \operatorname{Cov}(\bm{\theta}(t))
  = \frac{r^2 \sigma_{\theta,\bm{c}}^2}{4}\,
    A^{-1} \bigl(I - \exp(-4 E A t)\bigr).
\]


Hence asymptotically


\[
  \operatorname{Cov}(\bm{\theta})
  = \frac{r^2 \sigma_{\theta,\bm{c}}^2}{4}\,A^{-1},
\]


and


\[
  \bm{\theta} \sim \mathcal{N}\!\left(\bm{0},\,\frac{r^2 \sigma_{\theta,\bm{c}}^2}{4}\,A^{-1}\right).
\]



\begin{lemma}
Let $E$ be a positive–definite symmetric $n \times n$ matrix and


\[
  \bm{X} \sim \mathcal{N}(\bm{0}, E).
\]


Then


\[
  \bm{X}^\top E^{-1} \bm{X} \sim \chi^2_n.
\]


\end{lemma}

\begin{proof}
Write $E = P P^\top$ with $P$ invertible. Put $\bm{Y} = P^{-1} \bm{X}$. Then


\[
  \operatorname{Cov}(\bm{Y})
  = P^{-1} \operatorname{Cov}(\bm{X}) (P^{-1})^\top
  = P^{-1} E (P^{-1})^\top
  = P^{-1} P P^\top (P^{-1})^\top
  = I.
\]


Hence $\bm{Y} \sim \mathcal{N}(\bm{0}, I)$ and thus $\bm{Y}^\top \bm{Y} \sim \chi^2_n$. But


\[
  \bm{Y}^\top \bm{Y}
  = \bm{X}^\top (P^{-1})^\top P^{-1} \bm{X}
  = \bm{X}^\top E^{-1} \bm{X}.
\]


\end{proof}

From this and the asymptotic covariance we obtain that, up to scaling, $e(\bm{\theta})$ asymptotically follows a chi–square law, yielding the behaviour in (1) and (2) after translating via~\eqref{eq:f-e} and~\eqref{eq:sigma}.

\subsection{The case of diagonal Hessian}

Assume


\[
  e(\bm{\theta}) = -\sum_{j=1}^n \varepsilon_j \theta_j^2
\]


and that we are in the situation of Example~1.2 so that


\[
  \mathbb{E}(\bm{c}\bm{c}^\top) = \operatorname{diag}(c_0^2,\dots,c_{n-1}^2).
\]



Then Statement (3) refines as follows.

\begin{enumerate}
\setcounter{enumi}{3}
\item The expectation of $\theta_j(t)$, $j = 1,\dots,n$, converges towards
$\theta_j^\ast$ as


\[
  \mathbb{E}\bigl(\theta_j(t)\bigr)
  = \theta_j^\ast
    + \bigl(\theta_j(0) - \theta_j^\ast\bigr) \exp\!\Bigl(-\frac{t}{\lambda_j}\Bigr),
\]


with
\begin{equation}
  \lambda_j
  = \frac{C}{8 r c_j^2 \varepsilon_j}.
  \label{eq:lambda_j}
\end{equation}
\end{enumerate}

Each $\theta_j(t)$ is normally distributed, and


\[
  \operatorname{Var}\bigl(\theta_j(t)\bigr)
  = \frac{r(1-d)C}{8\varepsilon_j}
    \Bigl(1 - \exp\bigl(-8 r c_j^2 \varepsilon_j t / C\bigr)\Bigr).
\]



In particular,


\[
  \mathbb{E}\bigl(\theta_j(t)\bigr)
  = \exp\bigl(-4 r c_j^2 \varepsilon_j t / C\bigr)\,\theta_j(0),
\]


and


\[
  \mathbb{E}\bigl(e(\bm{\theta}(t))\bigr)
  = \sum_{j=1}^n \left(
      -\varepsilon_j \exp\bigl(-8 r c_j^2 \varepsilon_j t / C\bigr)\,\theta_j(0)^2
      + \frac{r(1-d)C}{8}\Bigl(1 - \exp\bigl(-8 r c_j^2 \varepsilon_j t / C\bigr)\Bigr)
    \right).
\]



Furthermore $e(\bm{\theta}(t))$ follows a generalized chi–square distribution~\cite{wiki:generalized-chi2}. The SPSA simulator~\cite{spsa_simul}
contains routines to compute the CDF and PPF for this distribution, based on MATLAB code by Abhranil Das~\cite{Das} and on Ruben's algorithm~\cite{Ruben}. Ruben's method expands the characteristic function $\varphi(t)$ as a power series in
$(1-2 i t \rho)^{-1/2}$ for suitable $\rho$, yielding a mixture of scaled $\chi^2_{n-2i}$–distributions for $i \ge 0$.

If


\[
  \exp\bigl(-8 r c_j^2 \varepsilon_j t / C\bigr)
  = \exp\bigl(-2t/\lambda_j\bigr)
\]


is small, then approximately


\[
  \frac{n\,e(\bm{\theta}(t))}{\mathbb{E}(e(\bm{\theta}(t)))} \sim \chi^2_n.
\]



Asymptotically,


\[
  \frac{e(\bm{\theta})}{\mathbb{E}(e(\bm{\theta}))} \sim \chi^2_n,
\]


so


\[
  \frac{n\,e(\bm{\theta}(0))}{\mathbb{E}(e(\bm{\theta}(0)))}
  \quad\text{and}\quad
  \frac{e(\bm{\theta}(t))}{\mathbb{E}(e(\bm{\theta}(t)))}
\]


act as scale factors to map asymptotic location quantities (means, percentiles, etc.) to their values at finite $t$ for $t \gg 0$.

If $c_j^2 \varepsilon_j$ is independent of $j$, we can set $\lambda = \lambda_j$ and obtain
\begin{equation}
  \mathbb{E}\bigl(e(\bm{\theta}(t))\bigr)
  = \mathbb{E}\bigl(e(\bm{\theta}(0))\bigr)\exp(-2t/\lambda)
    + \mathbb{E}\bigl(e(\bm{\theta})\bigr)\bigl(1-\exp(-2t/\lambda)\bigr).
  \label{eq:e-evolution}
\end{equation}

\subsection{Optimal approach}

Fix a target $-p < 0$. From~\eqref{eq:e-evolution} we choose $r$ so that $\mathbb{E}\bigl(e(\bm{\theta}(t))\bigr) = -p$ is achieved in minimal time $t$. We must solve


\[
  p = p_0 \exp(-a r t) + b r \bigl(1-\exp(-a r t)\bigr),
\]


with
\begin{equation}
  p_0 = -\mathbb{E}\bigl(e(\bm{\theta}(0))\bigr), \qquad
  a = \frac{8 c_j^2 \varepsilon_j}{C}, \qquad
  b = \frac{n(1-d) C}{8}.
  \label{eq:abc}
\end{equation}

After rescaling we obtain the universal equation
\begin{equation}
  \exp(-R T) + R\bigl(1-\exp(-R T)\bigr) = p,
\end{equation}
with


\[
  R = \frac{b r}{p_0}, \qquad
  p = \frac{p}{p_0}, \qquad
  T = a t.
\]



Using~\eqref{eq:abc}, one can express $T$ in terms of $R$ and $p$, then find $R$ such that $\partial T(R,p)/\partial R = 0$. This last step is done numerically. The optimal $R$ as a function of $p$ starts near $0$ at $p=0$, grows to a maximum, then decreases towards $0$ as $p \to 1$.
