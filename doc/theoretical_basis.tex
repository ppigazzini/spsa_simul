\section{The SPSA process}

We analyze an elegant chess-oriented version of the SPSA algorithm invented by
Stockfish author Joona Kiiski.

Let $(\bm{c}_i)_{i\ge 0}\subset\R^n$ be a sequence of column vectors and let
$(r_i)_{i\ge 0}\subset\R_{>0}$. To go from a vector of engine parameters

\[
  \bm{\theta}_i=(\theta_i^{(j)})_{j=1}^n\in\R^n
\]

to a new vector $\bm{\theta}_{i+1}\in\R^n$ we play a match of
$\bm{\theta}_i+\bm{c}_i$ against $\bm{\theta}_i-\bm{c}_i$ and set
\begin{equation}
  \bm{\theta}_{i+1} =
  \begin{cases}
    \bm{\theta}_i + r_i \bm{c}_i & \text{in case of a win},\\[4pt]
    \bm{\theta}_i                & \text{in case of a draw},\\[4pt]
    \bm{\theta}_i - r_i \bm{c}_i & \text{in case of a loss.}
  \end{cases}
  \label{eq:update}
\end{equation}

For exposition we assume games are played with a balanced opening book
(practically approximated by replaying games with reversed colors). This avoids
systematic opening bias.

\begin{example}\label{ex:fdsa}
Let $c_0,\dots,c_{n-1}>0$ and put

\[
  \bm{c}_i = c_{i\bmod n}\,\bm{e}_{i\bmod n},
\]

where $\bm{e}_j$ is the $j$th standard basis vector of $\R^n$. This is a
variant of FDSA. Then

\[
  \E(\bm{c}_i\bm{c}_i^\top)=\diag(c_0^2,\dots,c_{n-1}^2).
\]
\end{example}

\begin{example}\label{ex:spsa}
Let $c_0,\dots,c_{n-1}>0$ and put

\[
  \bm{c}_i=\sum_{j=0}^{n-1}\alpha_{i,j}c_j\bm{e}_j,
\]

where each $\alpha_{i,j}$ is chosen uniformly at random in $\{\pm1\}$. Then

\[
  \E(\bm{c}_i\bm{c}_i^\top)=\diag(c_0^2,\dots,c_{n-1}^2).
\]

This is the SPSA algorithm.
\end{example}

\begin{assumption}
We assume $(\bm{c}_i\bm{c}_i^\top)_i$ stays close to its short-term running
average $\E(\bm{c}_i\bm{c}_i^\top)$ and that this expectation is an invertible
$n\times n$ matrix.
\end{assumption}

\section{Main results}

\subsection{Statements}

Let $e$ map $\bm{\theta}\in\R^n$ to Elo. Assume $e$ is approximately quadratic
and attains its maximum at $\bm{\theta}^\ast$, normalized so
$e(\bm{\theta}^\ast)=0$.

Apply generalized SPSA with constant $r$ and with $\bm{c}_i$ satisfying
\Cref{ass:Ecct}.

\begin{assumption}\label{ass:Ecct}
For all $i$, $\E(\bm{c}_i\bm{c}_i^\top)$ exists, is constant in $i$, and is
invertible.
\end{assumption}

\begin{enumerate}
\item Asymptotically,
\begin{equation}
  \E\bigl(e(\bm{\theta}_i)\bigr)
  \approx -\frac{r}{8}\,C\,n(1-d),
  \label{eq:main1}
\end{equation}
where $d\in[0,1]$ is the draw ratio and

\[
  C=\frac{800}{\ln 10}\approx 347.43558552.
\]

\item Let $0\le\gamma\le1$ and define $z$ by $\gamma=\Prob(X<z)$ for
$X\sim\chi^2_n$. Then with probability $\gamma$,
\begin{equation}
  e(\bm{\theta}_i)\gtrsim -\frac{r}{8}\,C\,z(1-d).
  \label{eq:main2}
\end{equation}

\item The expectation $\E(\bm{\theta}_i)$ converges linearly to
$\bm{\theta}^\ast$ with time constant
\begin{equation}
  \lambda\approx\frac{C}{2r\mu},
  \label{eq:lambda}
\end{equation}
where $\mu$ is the smallest eigenvalue of

\[
  \E(\bm{c}_i\bm{c}_i^\top)\,\Hess(e)(\bm{\theta}^\ast).
\]
\end{enumerate}

\begin{remark}
If $\E(\bm{c}_i\bm{c}_i^\top)\,\Hess(e)(\bm{\theta}^\ast)=\mu I_n$, then the
average update is proportional to

\[
  r\,\Hess(e)(\bm{\theta}^\ast)^{-1}(\nabla e)(\bm{\theta}_i),
\]

the optimal descent direction.
\end{remark}

\subsection{Derivations}

Below vectors are column vectors. Switch to continuous time and write
$\bm{\theta}(t)$.

Model the discrete dynamics by the SDE
\begin{equation}
  d\bm{\theta}_t = r\bm{c}\bigl(\mu_{\theta,\bm{c}}\,dt + \sigma_{\theta,\bm{c}}\,dW_t\bigr),
\end{equation}
where $(\mu_{\theta,\bm{c}},\sigma_{\theta,\bm{c}})$ are mean and standard
deviation of a single game between $\bm{\theta}\pm\bm{c}$ (scored $-1,0,1$)
and $W_t$ is a Wiener process. Assume $\sigma_{\theta,\bm{c}}$ is independent
of $\bm{\theta},\bm{c}$ so
\begin{equation}
  \sigma_{\theta,\bm{c}}^2 = 1-d,
  \label{eq:sigma}
\end{equation}
with $d$ the draw ratio.

Let $f(\bm{\theta})$ be expected score against a reference engine. Assuming
linearity,

\[
  \mu_{\theta,\bm{c}} = \score(\bm{\theta}+\bm{c},\bm{\theta}-\bm{c})
  = f(\bm{\theta}+\bm{c})-f(\bm{\theta}-\bm{c}) = 2\bm{c}^\top(\nabla f)(\bm{\theta}),
\]

so
\begin{equation}
  d\bm{\theta}_t = 2r\bm{c}\bm{c}^\top(\nabla f)(\bm{\theta})\,dt + r\sigma_{\theta,\bm{c}}\bm{c}\,dW_t.
\end{equation}

Replace $\bm{c}\bm{c}^\top$ by its expectation and set
$A=r\E(\bm{c}\bm{c}^\top)$ (symmetric). Then
\begin{equation}
  d\bm{\theta}_t = 2A(\nabla f)(\bm{\theta})\,dt + r\sigma_{\theta,\bm{c}}\bm{c}\,dW_t.
\end{equation}

Using the approximate conversion
\begin{equation}
  f(\bm{\theta})=\frac{e(\bm{\theta})}{C},
  \label{eq:f-e}
\end{equation}
and writing $f(\bm{\theta})=-\tfrac12\bm{\theta}^\top E\bm{\theta}$ (with $E$ positive definite), we have $(\nabla f)(\bm{\theta})=-E\bm{\theta}$ and
\begin{equation}
  d\bm{\theta}_t = -2AE\bm{\theta}\,dt + r\sigma_{\theta,\bm{c}}\bm{c}\,dW_t.
\end{equation}

The homogeneous solution is $\bm{\theta}(t)=\exp(-2AE t)\bm{D}$. Variation of constants yields


\[
  \bm{\theta}(t)=\exp(-2AE t)\bm{\theta}(0) + r\sigma_{\theta,\bm{c}}\int_0^t \exp(2AE(s-t))\bm{c}\,dW_s.
\]


Hence

\[
  \E(\bm{\theta}(t))=\exp(-2AE t)\bm{\theta}(0)=\exp\bigl(-2r\E(\bm{c}\bm{c}^\top)E t\bigr)\bm{\theta}(0),
\]


which gives the linear convergence in expectation.

For the covariance, replacing $\bm{c}\bm{c}^\top$ by its expectation,
\begin{align*}
  \Cov(\bm{\theta}(t))
  &= r^2\sigma_{\theta,\bm{c}}^2\int_0^t \exp(2AE(s-t))\,\E(\bm{c}\bm{c}^\top)\,\exp(2EA(s-t))\,ds \\
  &= r^2\sigma_{\theta,\bm{c}}^2\int_0^t \exp\bigl(4EA(s-t)\bigr)\,A\,ds,
\end{align*}
and if $E$ and $A$ commute,


\[
  \Cov(\bm{\theta}(t))=\frac{r^2\sigma_{\theta,\bm{c}}^2}{4}A^{-1}\bigl(I-\exp(-4EA t)\bigr).
\]


Asymptotically,


\[
  \Cov(\bm{\theta})=\frac{r^2\sigma_{\theta,\bm{c}}^2}{4}A^{-1},
\]


so


\[
  \bm{\theta}\sim\mathcal{N}\!\left(\bm{0},\frac{r^2\sigma_{\theta,\bm{c}}^2}{4}A^{-1}\right).
\]



\begin{lemma}
Let $E$ be positive-definite symmetric and $\bm{X}\sim\mathcal{N}(\bm{0},E)$. Then $\bm{X}^\top E^{-1}\bm{X}\sim\chi^2_n$.
\end{lemma}

\begin{proof}
Write $E=PP^\top$ and set $\bm{Y}=P^{-1}\bm{X}$. Then $\Cov(\bm{Y})=I$, so $\bm{Y}^\top\bm{Y}\sim\chi^2_n$. But $\bm{Y}^\top\bm{Y}=\bm{X}^\top E^{-1}\bm{X}$.
\end{proof}

Combining the lemma with the asymptotic covariance yields the chi-square behaviour of $e(\bm{\theta})$ after scaling via~\eqref{eq:f-e} and~\eqref{eq:sigma}.

\subsection{Diagonal Hessian case}

Assume


\[
  e(\bm{\theta})=-\sum_{j=1}^n \varepsilon_j\theta_j^2,
\]


and $\E(\bm{c}\bm{c}^\top)=\diag(c_0^2,\dots,c_{n-1}^2)$ (\Cref{ex:fdsa}). Then each coordinate decouples and


\[
  \E(\theta_j(t))=\theta_j^\ast + (\theta_j(0)-\theta_j^\ast)\exp\!\Bigl(-\frac{t}{\lambda_j}\Bigr),
\]


with
\begin{equation}
  \lambda_j=\frac{C}{8r c_j^2\varepsilon_j}.
  \label{eq:lambda_j}
\end{equation}

Moreover,


\[
  \Var(\theta_j(t))=\frac{r(1-d)C}{8\varepsilon_j}\Bigl(1-\exp\bigl(-8r c_j^2\varepsilon_j t/C\bigr)\Bigr),
\]


and


\[
  \E(e(\bm{\theta}(t)))=\sum_{j=1}^n\Bigl(-\varepsilon_j\exp\bigl(-8r c_j^2\varepsilon_j t/C\bigr)\theta_j(0)^2 + \frac{r(1-d)C}{8}\bigl(1-\exp\bigl(-8r c_j^2\varepsilon_j t/C\bigr)\bigr)\Bigr).
\]



Asymptotically $e(\bm{\theta}(t))$ follows a generalized chi-square distribution; the SPSA simulator contains routines to compute its CDF and PPF (see references).

If $c_j^2\varepsilon_j$ is independent of $j$ (so $\lambda_j=\lambda$), then
\begin{equation}
  \E(e(\bm{\theta}(t)))=\E(e(\bm{\theta}(0)))\exp(-2t/\lambda)+\E(e(\bm{\theta}))\bigl(1-\exp(-2t/\lambda)\bigr).
  \label{eq:e-evolution}
\end{equation}

\subsection{Optimal approach}

Fix target $-p<0$. From~\eqref{eq:e-evolution} solve for $t$ given $r$:


\[
  p = p_0\exp(-a r t) + b r\bigl(1-\exp(-a r t)\bigr),
\]


with


\[
  p_0=-\E(e(\bm{\theta}(0))),\qquad a=\frac{8c_j^2\varepsilon_j}{C},\qquad b=\frac{n(1-d)C}{8}.
\]


Rescale to the universal equation


\[
  \exp(-RT)+R\bigl(1-\exp(-RT)\bigr)=p,
\]


with $R=\frac{br}{p_0}$, $T=at$. Solve numerically for $R$ minimizing $T(R,p)$.
